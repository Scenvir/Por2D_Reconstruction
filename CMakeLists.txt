cmake_minimum_required(VERSION 3.16)
project(por2_reconstructed LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(por2_core src/levels.cpp src/portal.cpp src/physics.cpp src/game.cpp src/render.cpp)
target_include_directories(por2_core PUBLIC include)
if(MSVC)
    target_compile_options(por2_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(por2_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_executable(por2 src/main.cpp)
    target_link_libraries(por2 PRIVATE por2_core gdi32 user32)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(por2_tests tests/tests.cpp tests/legacy_bridge.cpp)
    target_include_directories(por2_tests PRIVATE tests/legacy_stubs)
    target_link_libraries(por2_tests PRIVATE por2_core)
    add_test(NAME por2_regression COMMAND por2_tests)
endif()
