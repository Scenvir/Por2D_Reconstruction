cmake_minimum_required(VERSION 3.16)
project(por2_reconstructed LANGUAGES CXX)
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_library(por2_core src/levels.cpp src/editor_level.cpp src/portal.cpp src/physics.cpp src/game.cpp src/render.cpp src/tutorial.cpp)
target_include_directories(por2_core PUBLIC include)
if(MSVC)
    target_compile_options(por2_core PRIVATE /W4)
else()
    target_compile_options(por2_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_executable(Por2D src/main.cpp)
    target_link_libraries(Por2D PRIVATE por2_core gdi32 user32 comdlg32 shell32 ole32 uuid)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(por2_tests tests/tests.cpp tests/legacy_bridge.cpp)
    target_include_directories(por2_tests PRIVATE tests/legacy_stubs)
    target_link_libraries(por2_tests PRIVATE por2_core)
    add_test(NAME por2_regression COMMAND por2_tests)
endif()
